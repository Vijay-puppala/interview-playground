@@ Testing basics

@@@ Table-driven checks
Run the same check over many inputs: the idea behind parametrized tests.
@py
cases = [("abc", 3), ("", 0), ("hello", 5)]
for text, expected in cases:
    assert len(text) == expected, f"{text!r}: expected {expected}"
print("all passed")
=> all passed
@js
const cases = [["abc", 3], ["", 0], ["hello", 5]];
for (const [text, expected] of cases) {
  if (text.length !== expected) throw new Error(`${JSON.stringify(text)}: expected ${expected}`);
}
console.log("all passed");
=> all passed

Use a test runner for discovery, reporting and fixtures.

import pytest

@@@ Fixtures: setup and teardown
@py!
import pytest

@pytest.fixture
def account():
    acct = {"balance": 100}      # setup
    yield acct                   # the test runs here
    acct.clear()                 # teardown

def test_withdraw(account):
    account["balance"] -= 30
    assert account["balance"] == 70
@js!
import test, { beforeEach, afterEach } from "node:test";
import assert from "node:assert/strict";

let account;
beforeEach(() => { account = { balance: 100 }; });   // setup
afterEach(() => { account = null; });                // teardown

test("withdraw", () => {
  account.balance -= 30;
  assert.equal(account.balance, 70);
});

@@@ Expect an exception
@py
def expect_raises(exc, fn):
    try:
        fn()
    except exc:
        return True
    return False

print(expect_raises(ZeroDivisionError, lambda: 1 / 0), expect_raises(ValueError, lambda: 1))
=> True False
@js
const expectThrows = (fn) => {
  try {
    fn();
  } catch {
    return true;
  }
  return false;
};
console.log(expectThrows(() => JSON.parse("{bad")), expectThrows(() => 1));
=> true false

@@@ Retry a flaky call
Retries hide flakiness: use them sparingly and fix the root cause.
@py
import functools

def retry(times=3):
    def decorator(fn):
        @functools.wraps(fn)
        def wrapper(*args, **kwargs):
            for attempt in range(1, times + 1):
                try:
                    return fn(*args, **kwargs)
                except Exception:
                    if attempt == times:
                        raise
        return wrapper
    return decorator

calls = {"n": 0}

@retry(times=3)
def flaky():
    calls["n"] += 1
    if calls["n"] < 3:
        raise ConnectionError("boom")
    return "ok"

print(flaky(), calls["n"])
=> ok 3
@js
const retry = (fn, times = 3) => async (...args) => {
  for (let attempt = 1; ; attempt++) {
    try {
      return await fn(...args);
    } catch (err) {
      if (attempt >= times) throw err;
    }
  }
};

let calls = 0;
const flaky = retry(async () => {
  if (++calls < 3) throw new Error("boom");
  return "ok";
});
console.log(await flaky(), calls);
=> ok 3

@@@ Diff expected and actual data
@py
expected = {"id": 1, "status": "active", "plan": "pro"}
actual = {"id": 1, "status": "inactive", "plan": "pro"}
diff = {k: (expected[k], actual.get(k)) for k in expected if expected[k] != actual.get(k)}

=> {'status': ('active', 'inactive')}
@js
const expected = { id: 1, status: "active", plan: "pro" };
const actual = { id: 1, status: "inactive", plan: "pro" };
const diff = Object.fromEntries(
  Object.keys(expected).filter((k) => expected[k] !== actual[k]).map((k) => [k, [expected[k], actual[k]]]),
);
console.log(JSON.stringify(diff));
=> {"status":["active","inactive"]}

@@@ Browser test with Selenium
@py!
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

driver = webdriver.Chrome()
try:
    driver.get("https://example.com")
    link = WebDriverWait(driver, 10).until(EC.element_to_be_clickable((By.LINK_TEXT, "More information")))
    link.click()
    assert "iana.org" in driver.current_url
finally:
    driver.quit()
@js!
const { Builder, By, until } = require("selenium-webdriver");

(async () => {
  const driver = await new Builder().forBrowser("chrome").build();
  try {
    await driver.get("https://example.com");
    const link = await driver.wait(until.elementLocated(By.linkText("More information")), 10000);
    await link.click();
    if (!(await driver.getCurrentUrl()).includes("iana.org")) throw new Error("wrong page");
  } finally {
    await driver.quit();
  }
})();

@@@ Page Object pattern
Keep locators and actions in one class so UI changes are fixed in one place.
@py!
class LoginPage:
    def __init__(self, page):
        self.page = page
        self.email = page.get_by_label("Email")
        self.password = page.get_by_label("Password")
        self.submit = page.get_by_role("button", name="Sign in")

    def login(self, email, password):
        self.email.fill(email)
        self.password.fill(password)
        self.submit.click()
        return self

def test_login(page):
    LoginPage(page).login("qa@example.com", "secret")
    expect(page.get_by_text("Welcome")).to_be_visible()
@js!
export class LoginPage {
  constructor(page) {
    this.page = page;
    this.email = page.getByLabel("Email");
    this.password = page.getByLabel("Password");
    this.submit = page.getByRole("button", { name: "Sign in" });
  }

  async login(email, password) {
    await this.email.fill(email);
    await this.password.fill(password);
    await this.submit.click();
    return this;
  }
}
@ts!
import type { Page, Locator } from "@playwright/test";

export class LoginPage {
  readonly email: Locator;
  readonly password: Locator;
  readonly submit: Locator;

  constructor(private readonly page: Page) {
    this.email = page.getByLabel("Email");
    this.password = page.getByLabel("Password");
    this.submit = page.getByRole("button", { name: "Sign in" });
  }

  async login(email: string, password: string): Promise<this> {
    await this.email.fill(email);
    await this.password.fill(password);
    await this.submit.click();
    return this;
  }
}
