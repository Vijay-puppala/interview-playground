@@ Async and concurrency

@@@ async / await and running tasks together
Wait for several slow operations (API calls, page loads) at the same time.
@py
import asyncio

async def fetch(n):
    await asyncio.sleep(0.01)
    return n * 2

async def main():
    print(await asyncio.gather(fetch(1), fetch(2)))

asyncio.run(main())
=> [2, 4]
@js
const fetchN = async (n) => {
  await new Promise((resolve) => setTimeout(resolve, 10));
  return n * 2;
};
console.log(JSON.stringify(await Promise.all([fetchN(1), fetchN(2)])));
=> [2,4]
! The browser cannot block, so this page's runner adapts asyncio.run() for you. In a real script it works exactly as written.

@@@ Timeouts
Give up on an operation that takes too long.
@py
import asyncio

async def slow():
    await asyncio.sleep(1)

async def main():
    try:
        await asyncio.wait_for(slow(), timeout=0.01)
    except asyncio.TimeoutError:
        print("timeout")

asyncio.run(main())
=> timeout
@js
const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));
const withTimeout = (p, ms) =>
  Promise.race([p, new Promise((_, reject) => setTimeout(() => reject(new Error("timeout")), ms))]);
try {
  await withTimeout(sleep(1000), 10);
} catch (e) {
  console.log(e.message);
}
=> timeout

@@@ Collect results even when some fail
@py
import asyncio

async def ok():
    return "ok"

async def bad():
    raise ValueError("boom")

async def main():
    results = await asyncio.gather(ok(), bad(), return_exceptions=True)
    print([type(r).__name__ if isinstance(r, Exception) else r for r in results])

asyncio.run(main())
=> ['ok', 'ValueError']
@js
const results = await Promise.allSettled([Promise.resolve("ok"), Promise.reject(new Error("boom"))]);
console.log(JSON.stringify(results.map((r) => (r.status === "fulfilled" ? r.value : "Error"))));
=> ["ok","Error"]

@@@ Poll until a condition is true (explicit wait)
The idea behind every Selenium / Playwright wait: check repeatedly, give up after a deadline.
@py
import time

attempts = {"n": 0}

def ready():
    attempts["n"] += 1
    return attempts["n"] >= 3

deadline = time.monotonic() + 1
while not ready():
    if time.monotonic() > deadline:
        raise TimeoutError("not ready")
    time.sleep(0.01)
print("ready after", attempts["n"], "tries")
=> ready after 3 tries
@js
let attempts = 0;
const ready = () => ++attempts >= 3;
const deadline = Date.now() + 1000;
while (!ready()) {
  if (Date.now() > deadline) throw new Error("not ready");
  await new Promise((resolve) => setTimeout(resolve, 10));
}
console.log("ready after", attempts, "tries");
=> ready after 3 tries
! Prefer waiting for a condition over a fixed sleep: fixed sleeps make tests slow and flaky.

@py!
from concurrent.futures import ThreadPoolExecutor

def check(url):

with ThreadPoolExecutor(max_workers=4) as pool:

@@ Types and typing

@@@ Type annotations
Python hints and TypeScript types describe the data; tools check them before the code runs.
@py
def add(a: int, b: int) -> int:
    return a + b

print(add(2, 3))
=> 5
@js
/**
 * @param {number} a
 * @param {number} b
 * @returns {number}
 */
function add(a, b) {
  return a + b;
}
console.log(add(2, 3));
=> 5
@ts
function add(a: number, b: number): number {
  return a + b;
}
console.log(add(2, 3));
=> 5
! Python ignores type hints at run time: use mypy or pyright to check them.

@@@ Union and optional types
A value that may be missing.
@py
def label(x: int | None) -> str:
    return "none" if x is None else str(x)

print(label(None), label(5))
=> none 5
@js
const label = (x) => x ?? "none";
console.log(label(null), label(5));
=> none 5
@ts
function label(x: number | null): string {
  return x === null ? "none" : String(x);
}
console.log(label(null), label(5));
=> none 5

@@@ Generics
One function that keeps the type of whatever you give it.
@py
from typing import TypeVar

T = TypeVar("T")

def first(items: list[T]) -> T | None:
    return items[0] if items else None

print(first([3, 4]), first([]))
=> 3 None
@js
const first = (items) => (items.length ? items[0] : null);
console.log(first([3, 4]), first([]));
=> 3 null
@ts
function first<T>(items: T[]): T | undefined {
  return items[0];
}
console.log(first([3, 4]), first<string>([]));
=> 3 undefined

@@@ Narrowing types
Check the type, then use the value safely.
@py
def size(x):
    return len(x) if isinstance(x, (str, list)) else x

print(size("abc"), size([1, 2]), size(5))
=> 3 2 5
@js
const size = (x) => (typeof x === "string" || Array.isArray(x) ? x.length : x);
console.log(size("abc"), size([1, 2]), size(5));
=> 3 2 5
@ts
function size(x: string | number[] | number): number {
  if (typeof x === "number") return x;   // narrowed to number here
  return x.length;                       // string | number[] both have length
}
console.log(size("abc"), size([1, 2]), size(5));
=> 3 2 5

@@@ Pick, omit and partial updates
@py
user = {"id": 1, "name": "Vijay", "age": 30}
picked = {k: user[k] for k in ("id", "name")}
omitted = {k: v for k, v in user.items() if k != "age"}
print(picked, omitted == picked)
=> {'id': 1, 'name': 'Vijay'} True
@js
const user = { id: 1, name: "Vijay", age: 30 };
const { id, name } = user;                 // pick
const { age, ...omitted } = user;          // omit one field
console.log(JSON.stringify({ id, name }), JSON.stringify(omitted));
=> {"id":1,"name":"Vijay"} {"id":1,"name":"Vijay"}
@ts
interface User { id: number; name: string; age: number }
type Public = Pick<User, "id" | "name">;       // keep some fields
type NoAge = Omit<User, "age">;                // drop a field
type Patch = Partial<User>;                    // every field optional
const patch: Patch = { name: "QA" };
const pub: Public = { id: 1, name: "Vijay" };
console.log(JSON.stringify(patch), JSON.stringify(pub));
=> {"name":"QA"} {"id":1,"name":"Vijay"}

@@@ Seeded random test data
A fixed seed makes "random" data reproducible, so a failing test can be replayed.
@py
import random
random.seed(42)
print(random.sample(range(10), 3), random.randint(1, 100))
=> [1, 0, 4] 32
@js
function mulberry32(seed) {     // tiny seeded generator (JavaScript has no built-in one)
  return () => {
    seed = (seed + 0x6d2b79f5) | 0;
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
const rand = mulberry32(42);
console.log(Math.floor(rand() * 100), Math.floor(rand() * 100), Math.floor(rand() * 100));
=> 60 44 85
! The same seed gives different numbers in different languages: the guarantee is repeatability within one language.
