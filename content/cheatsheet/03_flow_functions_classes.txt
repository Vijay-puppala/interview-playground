@@ Control flow and loops

@@@ if / elif / else
@py
score = 82
if score >= 90:
    grade = "A"
elif score >= 75:
    grade = "B"
else:
    grade = "C"
print(grade)
=> B
@js
const score = 82;
let grade;
if (score >= 90) grade = "A";
else if (score >= 75) grade = "B";
else grade = "C";
console.log(grade);
=> B

@@@ for loops and range

@@@ while, break and continue
@py
out = []
n = 0
while True:
    n += 1
    if n % 2 == 0:
        continue          # skip even numbers
    if n > 7:
        break             # stop the loop
    out.append(n)
print(out)
=> [1, 3, 5, 7]
@js
const out = [];
let n = 0;
while (true) {
  n += 1;
  if (n % 2 === 0) continue;   // skip even numbers
  if (n > 7) break;            // stop the loop
  out.push(n);
}
console.log(JSON.stringify(out));
=> [1,3,5,7]

@@@ match / switch
Branch on a value. Python 3.10+ has structural pattern matching.
@py
def kind(code):
    match code:
        case 200 | 201:
            return "ok"
        case 404:
            return "missing"
        case _:
            return "other"

print(kind(201), kind(404), kind(500))
=> ok missing other
@js
function kind(code) {
  switch (code) {
    case 200:
    case 201:
      return "ok";
    case 404:
      return "missing";
    default:
      return "other";
  }
}
console.log(kind(201), kind(404), kind(500));
=> ok missing other
! JavaScript switch falls through to the next case unless you break or return.

@@@ Variable arguments
Accept any number of arguments.
@py
def total(*nums, **opts):
    return sum(nums) * opts.get("factor", 1)

print(total(1, 2, 3), total(1, 2, factor=10))
=> 6 30
@js
function total(factor, ...nums) {
  return factor * nums.reduce((a, b) => a + b, 0);
}
console.log(total(1, 1, 2, 3), total(10, 1, 2));
=> 6 30
@ts
function total(factor: number, ...nums: number[]): number {
  return factor * nums.reduce((a, b) => a + b, 0);
}
console.log(total(1, 1, 2, 3), total(10, 1, 2));
=> 6 30

@@@ Closures
A function that remembers variables from where it was created.
@py
def counter():
    count = 0
    def inc():
        nonlocal count
        count += 1
        return count
    return inc

c = counter()
print(c(), c(), c())
=> 1 2 3
@js
function counter() {
  let count = 0;
  return () => ++count;
}
const c = counter();
console.log(c(), c(), c());
=> 1 2 3

@@@ Decorators and wrapper functions
Add behaviour (logging, timing, retries) around a function without changing it.
@py
def logged(fn):
    def wrapper(*args, **kwargs):
        result = fn(*args, **kwargs)
        print(f"{fn.__name__} done")
        return result
    return wrapper

@logged
def work():
    return 42

print(work())
=> work done
=> 42
@js
const logged = (fn) => (...args) => {
  const result = fn(...args);
  console.log(`${fn.name} done`);
  return result;
};
const work = logged(function work() { return 42; });
console.log(work());
=> work done
=> 42

@@@ Generators
Produce values lazily, one at a time.
@py
def countdown(n):
    while n > 0:
        yield n
        n -= 1

print(list(countdown(3)))
=> [3, 2, 1]
@js
function* countdown(n) {
  while (n > 0) yield n--;
}
console.log(JSON.stringify([...countdown(3)]));
=> [3,2,1]

def stats(nums):

@@@ Recursion
@py
def fact(n):
    return 1 if n <= 1 else n * fact(n - 1)

print(fact(5))
=> 120
@js
const fact = (n) => (n <= 1 ? 1 : n * fact(n - 1));

! Python stops at about 1000 nested calls (RecursionError). Prefer a loop for deep recursion.

@@ Classes and OOP

@@@ Class basics
@py
class Account:
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.balance = balance

    def deposit(self, amount):
        self.balance += amount
        return self

    def __repr__(self):
        return f"Account({self.owner!r}, {self.balance})"

print(Account("Vijay").deposit(50))
=> Account('Vijay', 50)
@js
class Account {
  constructor(owner, balance = 0) {
    this.owner = owner;
    this.balance = balance;
  }
  deposit(amount) {
    this.balance += amount;
    return this;
  }
  toString() {
    return `Account(${this.owner}, ${this.balance})`;
  }
}
console.log(String(new Account("Vijay").deposit(50)));
=> Account(Vijay, 50)
@ts
class Account {
  constructor(public owner: string, public balance: number = 0) {}
  deposit(amount: number): this {
    this.balance += amount;
    return this;
  }
  toString(): string {
    return `Account(${this.owner}, ${this.balance})`;
  }
}
console.log(String(new Account("Vijay").deposit(50)));
=> Account(Vijay, 50)

@@@ Properties and static methods
@py
class Temp:
    def __init__(self, c):
        self.c = c

    @property
    def f(self):
        return self.c * 9 / 5 + 32

    @staticmethod
    def zero():
        return Temp(0)

print(Temp(100).f, Temp.zero().f)
=> 212.0 32.0
@js
class Temp {

@@@ Abstract classes and interfaces
Force subclasses to implement a method.
@py
from abc import ABC, abstractmethod

class Shape(ABC):
    @abstractmethod
    def area(self): ...

class Square(Shape):
    def __init__(self, s):
        self.s = s
    def area(self):
        return self.s ** 2

print(Square(3).area())
try:
    Shape()
except TypeError:
    print("abstract")
=> 9
=> abstract
@js
class Shape {
  constructor() {
    if (new.target === Shape) throw new TypeError("abstract");
  }
}
class Square extends Shape {
  constructor(s) { super(); this.s = s; }
  area() { return this.s ** 2; }
}
console.log(new Square(3).area());
try {
  new Shape();
} catch (e) {
  console.log("abstract");
}
=> 9
=> abstract
@ts
abstract class Shape {
  abstract area(): number;
}
class Square extends Shape {
  constructor(private s: number) { super(); }
  area(): number { return this.s ** 2; }
}
console.log(new Square(3).area());
=> 9

@@@ Magic methods and equality
@py
class Money:
    def __init__(self, cents):
        self.cents = cents
    def __eq__(self, other):
        return self.cents == other.cents
    def __add__(self, other):
        return Money(self.cents + other.cents)

print(Money(150) + Money(50) == Money(200))
=> True
@js
class Money {
  constructor(cents) { this.cents = cents; }
  equals(other) { return this.cents === other.cents; }
  plus(other) { return new Money(this.cents + other.cents); }
}
console.log(new Money(150).plus(new Money(50)).equals(new Money(200)));
=> true
! JavaScript has no operator overloading: use methods such as equals() and plus().

@@ Errors and exceptions

@@@ try / except / finally
@py
def parse(s):
    try:
        return int(s)
    except ValueError:
        return f"bad: {s}"
    finally:
        print("checked", s)

print(parse("7"), parse("x"))
=> checked 7
=> checked x
=> 7 bad: x
@js
function parse(s) {
  try {
    return JSON.parse(s);
  } catch (e) {
    return `bad: ${s}`;
  } finally {
    console.log("checked", s);
  }
}
console.log(parse("7"), parse("x"));
=> checked 7
=> checked x
=> 7 bad: x

@@@ Raise and custom errors
@py
class ValidationError(Exception):
    pass

def check_age(age):
    if age < 0:
        raise ValidationError(f"invalid age: {age}")
    return age

try:
    check_age(-1)
except ValidationError as e:
    print(type(e).__name__, e)
=> ValidationError invalid age: -1
@js
class ValidationError extends Error {
  constructor(message) {
    super(message);
    this.name = "ValidationError";
  }
}
function checkAge(age) {
  if (age < 0) throw new ValidationError(`invalid age: ${age}`);
  return age;
}
try {
  checkAge(-1);
} catch (e) {
  console.log(e.name, e.message);
}
=> ValidationError invalid age: -1

@@@ Assertions
The building block of every automated test.
@py
try:

@js
function assert(condition, message) {
  if (!condition) throw new Error(message);
}
try {
  assert(2 + 2 === 5, "math is broken");
} catch (e) {
  console.log("failed:", e.message);
}
=> failed: math is broken

@@@ Clean up resources (with / finally)

const resource = { open: () => console.log("open"), close: () => console.log("close") };

} finally {

}

=> work
=> close
